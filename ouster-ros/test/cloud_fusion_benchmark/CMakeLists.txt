cmake_minimum_required(VERSION 3.18)
project(cloud_fusion_benchmark LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(required BASELINE_ROS_SOURCE CANDIDATE_ROS_SOURCE FIXTURE_HELPERS SDK_SOURCE SDK_BUILD)
  if(NOT ${required})
    message(FATAL_ERROR "Specify ${required}")
  endif()
endforeach()
find_package(ament_cmake REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(pcl_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(message_filters REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
find_library(ZIP_LIBRARY NAMES zip REQUIRED)
find_path(PCL_CONVERSIONS_INCLUDE_DIR pcl_conversions/pcl_conversions.h
  HINTS "${pcl_conversions_DIR}/../../../include" PATH_SUFFIXES pcl_conversions REQUIRED)
add_library(sdk STATIC IMPORTED)
set_target_properties(sdk PROPERTIES
  IMPORTED_LOCATION "${SDK_BUILD}/ouster_client/libouster_client.a"
  INTERFACE_INCLUDE_DIRECTORIES "${SDK_SOURCE}/ouster_client/include;${SDK_SOURCE}/ouster_client/include/optional-lite"
  INTERFACE_LINK_LIBRARIES "${SDK_BUILD}/ouster_client/nmea/libnmea.a;${ZIP_LIBRARY};OpenSSL::SSL;OpenSSL::Crypto;CURL::libcurl;Threads::Threads")
add_library(baseline OBJECT variant.cpp)
# Distinct namespaces prevent ODR interposition between frozen and candidate
# template bodies, while retaining identical point layouts and compiler flags.
target_compile_definitions(baseline PRIVATE BASELINE ouster_ros=ouster_ros_baseline)
add_library(candidate OBJECT variant.cpp)
foreach(target baseline candidate)
  if(target STREQUAL "baseline")
    set(source_root "${BASELINE_ROS_SOURCE}")
  else()
    set(source_root "${CANDIDATE_ROS_SOURCE}")
  endif()
  target_include_directories(${target} PRIVATE "${source_root}/src" "${source_root}/include")
endforeach()
add_executable(cloud_fusion_benchmark main.cpp $<TARGET_OBJECTS:baseline> $<TARGET_OBJECTS:candidate>)
foreach(target baseline candidate cloud_fusion_benchmark)
  target_include_directories(${target} PRIVATE "${FIXTURE_HELPERS}" ${PCL_INCLUDE_DIRS} ${PCL_CONVERSIONS_INCLUDE_DIR})
  target_link_libraries(${target} sdk Eigen3::Eigen pcl_common ${sensor_msgs_TARGETS}
    ${pcl_msgs_TARGETS} rclcpp::rclcpp message_filters::message_filters)
  target_compile_options(${target} PRIVATE -O3 -DNDEBUG)
endforeach()
